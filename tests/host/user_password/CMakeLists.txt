cmake_minimum_required(VERSION 3.16)
project(betta_password_test C)
if(NOT IDF_PATH)
    set(IDF_PATH "$ENV{IDF_PATH}")
endif()
if(NOT EXISTS "${IDF_PATH}/components/mbedtls/mbedtls/tf-psa-crypto/CMakeLists.txt")
    message(FATAL_ERROR "Set IDF_PATH to the existing ESP-IDF 6.1 checkout")
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(TF_PSA_CRYPTO_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/crypto_config.h" CACHE FILEPATH "" FORCE)
add_subdirectory("${IDF_PATH}/components/mbedtls/mbedtls/tf-psa-crypto" psa)
add_executable(test_user_password ../test_user_password.c ../../../main/settings/user_password.c)
target_include_directories(test_user_password PRIVATE ../../../main)
target_link_libraries(test_user_password PRIVATE tfpsacrypto)
