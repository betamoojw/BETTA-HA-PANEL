cmake_minimum_required(VERSION 3.16)
project(betta_knx_tests LANGUAGES C CXX)
enable_testing()
add_compile_options(-UNDEBUG -Wall -Wextra)
set(CMAKE_CXX_STANDARD 17)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(KNX "${ROOT}/main/knx")
if(NOT ESP_KNX_IP_DIR)
  set(ESP_KNX_IP_DIR "${ROOT}/managed_components/esp_knx_ip")
endif()
if(NOT CJSON_DIR)
  set(CJSON_DIR "${ROOT}/managed_components/espressif__cjson/cJSON")
endif()
add_library(knx_codec STATIC "${KNX}/knx_object.cc" "${KNX}/knx_validation.cc"
  "${KNX}/knx_config.cc" "${KNX}/knx_json.cc"
  "${ESP_KNX_IP_DIR}/src/knx_dpt.cpp" "${CJSON_DIR}/cJSON.c")
target_include_directories(knx_codec PUBLIC "${KNX}" "${ESP_KNX_IP_DIR}/include" "${CJSON_DIR}")
add_executable(knx_codec_test test_knx_object.cc test_knx_validation.cc)
target_link_libraries(knx_codec_test PRIVATE knx_codec)
add_test(NAME knx_codec COMMAND knx_codec_test)
add_executable(knx_config_test test_config.cc)
target_link_libraries(knx_config_test PRIVATE knx_codec)
add_test(NAME knx_config COMMAND knx_config_test)
add_executable(knx_mcp_test test_mcp.c "${KNX}/knx_mcp.c" "${CJSON_DIR}/cJSON.c")
target_include_directories(knx_mcp_test PRIVATE "${ROOT}/main" "${ROOT}/tests/host/stubs" "${CJSON_DIR}" stubs)
add_test(NAME knx_mcp COMMAND knx_mcp_test)
add_executable(xiaozhi_mcp_network_test test_mcp_network.c
  "${ROOT}/main/xiaozhi/xiaozhi_mcp_network.c" "${CJSON_DIR}/cJSON.c")
target_include_directories(xiaozhi_mcp_network_test PRIVATE stubs/network stubs "${ROOT}/tests/host/stubs" "${ROOT}/main" "${CJSON_DIR}")
add_test(NAME xiaozhi_mcp_network COMMAND xiaozhi_mcp_network_test)
add_executable(xiaozhi_mcp_ha_test test_mcp_ha.c
  "${ROOT}/main/xiaozhi/xiaozhi_mcp_ha.c" "${ROOT}/main/ha/ha_model.c"
  "${ROOT}/main/ha/ha_layout_entities.c" "${CJSON_DIR}/cJSON.c")
target_sources(xiaozhi_mcp_ha_test PRIVATE "${ROOT}/main/layout/bambu_printer_slots.c" "${ROOT}/main/layout/evcc_slots.c"
  "${ROOT}/main/xiaozhi/xiaozhi_ha_control.c" "${ROOT}/main/ha/ha_light_capabilities.c")
target_sources(xiaozhi_mcp_ha_test PRIVATE "${ROOT}/main/ha/ha_service_request.c")
target_include_directories(xiaozhi_mcp_ha_test PRIVATE stubs/ha "${ROOT}/tests/host/stubs" "${ROOT}/main" "${CJSON_DIR}")
add_test(NAME xiaozhi_mcp_ha COMMAND xiaozhi_mcp_ha_test)
# ESP-IDF archives main's source objects. A weak fallback in the dispatcher can
# satisfy its own references and prevent extraction of the real device adapter.
add_library(knx_mcp_link_archive STATIC "${KNX}/knx_mcp.c" mcp_link_adapter.c)
target_include_directories(knx_mcp_link_archive PRIVATE "${ROOT}/main" "${ROOT}/tests/host/stubs" "${CJSON_DIR}" stubs)
add_executable(knx_mcp_link_test test_mcp_link.c "${CJSON_DIR}/cJSON.c")
target_include_directories(knx_mcp_link_test PRIVATE "${ROOT}/main" "${ROOT}/tests/host/stubs" "${CJSON_DIR}" stubs)
target_link_libraries(knx_mcp_link_test PRIVATE knx_mcp_link_archive)
add_test(NAME knx_mcp_link COMMAND knx_mcp_link_test)
add_executable(knx_protocol_test "${ESP_KNX_IP_DIR}/test/host/test_protocol.cpp"
  "${ESP_KNX_IP_DIR}/src/knx_ip_protocol.cpp" "${ESP_KNX_IP_DIR}/src/knx_dpt.cpp")
target_include_directories(knx_protocol_test PRIVATE "${ESP_KNX_IP_DIR}/include" "${ESP_KNX_IP_DIR}/private_include")
add_test(NAME knx_protocol COMMAND knx_protocol_test)
add_executable(knx_service_test test_service.cc "${KNX}/knx_service.cc")
target_include_directories(knx_service_test BEFORE PRIVATE stubs "${ROOT}/tests/host/stubs" "${ROOT}/main")
target_link_libraries(knx_service_test PRIVATE knx_codec)
add_test(NAME knx_service COMMAND knx_service_test)
add_executable(knx_api_test test_api.c "${KNX}/knx_api.c" "${CJSON_DIR}/cJSON.c")
target_include_directories(knx_api_test PRIVATE "${ROOT}/main" "${ROOT}/tests/host/stubs" "${CJSON_DIR}" stubs)
add_test(NAME knx_api COMMAND knx_api_test)
